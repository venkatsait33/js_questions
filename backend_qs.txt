npm init -y to create a package.json file
packages installed are express, cors, dotenv, mongoose, bcrypt, jsonwebtoken, nodemailer, express-validator, multer, and nodemon

create a index.js file and import express, cors, dotenv, mongoose, and bcrypt and start app server.

create a .env file and add the following variables:
PORT=3000
DB_URL=mongodb+srv://<username>:<password>@cluster0.9yq6v.mongodb.net/<dbname>?retryWrites=true&w=majority
JWT_SECRET=<jwt_secret>
EMAIL_HOST=<email_host>
EMAIL_PORT=<email_port>
EMAIL_USER=<email_user>
EMAIL_PASS=<email_password>

in index.js,  we create const app = express() 
app.listen(process.env.PORT, () => console.log(`Server is running on port ${process.env.PORT}`))
app.cookieParser(jsonwebtoken.sign({userId: user._id}, process.env.JWT_SECRET, {expiresIn: '1h'}))
app.use(cors())
app.use(express.json())
app.use(express.urlencoded({extended: true}))
app.use(cookieParser())
app.use(express.static('public'))
